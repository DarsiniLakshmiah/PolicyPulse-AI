```python
def fetch_comments(docket_id, max_comments=50):
    """Pull comments from Regulations.gov"""
    url = f"{REGULATIONS_API}/comments"
    params = {
        "filter[docketId]": docket_id,
        "api_key": API_KEY,
        "page[size]": max_comments,
        "fields[comments]": "id,attributes"
    }
    try:
        r = requests.get(url, params=params, timeout=10)
        data = r.json()
        comments = []
        for item in data.get("data", []):
            text = item.get("attributes", {}).get("comment", "")
            comment_id = item.get("id", "")
            if text and len(text) > 20:
                comments.append({"id": comment_id, "text": text, "docket": docket_id})
        return comments
    except Exception as e:
        print(f"API error for {docket_id}: {e}")
        return []

def fetch_preamble_text(docket_id):
    """Pull final rule preamble from Federal Register"""
    # Search Federal Register for final rule matching docket
    url = f"{FEDERAL_REGISTER_API}/documents.json"
    params = {
        "conditions[docket_id]": docket_id,
        "conditions[type][]": "Rule",
        "fields[]": ["full_text_xml_url", "document_number"]
    }
    try:
        r = requests.get(url, params=params, timeout=10)
        docs = r.json().get("results", [])
        if not docs:
            return ""
        # Fetch full text of most recent final rule
        doc_num = docs[0].get("document_number", "")
        text_url = f"{FEDERAL_REGISTER_API}/documents/{doc_num}.json"
        r2 = requests.get(text_url, timeout=10)
        return r2.json().get("body_html_url", "")
    except Exception as e:
        print(f"Preamble fetch error: {e}")
        return ""

def auto_label(comment_id, preamble_text):
    """
    Label = 1 (significant) if comment ID appears in the preamble
    This is the core innovation: preamble = ground truth for what agency responded to
    """
    if not preamble_text:
        return None  # Can't label without preamble
    # Comment codes appear in preambles like "EPA-HQ-OAR-2021-0257-0143"
    return 1 if comment_id in preamble_text else 0

def train_model(training_data):
    """Train TF-IDF + LogisticRegression pipeline"""
    texts = [d["text"] for d in training_data]
    labels = [d["significant"] for d in training_data]

    pipeline = Pipeline([
        ("tfidf", TfidfVectorizer(
            ngram_range=(1, 2),
            max_features=5000,
            stop_words="english",
            min_df=1
        )),
        ("clf", LogisticRegression(
            C=1.0,
            class_weight="balanced",
            max_iter=1000,
            random_state=42
        ))
    ])

    if len(set(labels)) > 1 and len(texts) > 4:
        X_train, X_test, y_train, y_test = train_test_split(
            texts, labels, test_size=0.2, random_state=42
        )
        pipeline.fit(X_train, y_train)
        y_pred = pipeline.predict(X_test)
        print("\nModel performance:")
        print(classification_report(y_test, y_pred,
              target_names=["Not Significant", "Significant"]))
    else:
        pipeline.fit(texts, labels)

    return pipeline

if __name__ == "__main__":

    print("Model saved to model/significance_clf.pkl")
    print("\nDone. Now run: python app.py")
```

## Step 2 — ml_scorer.py
```python
"""
ML-based comment scoring. Loads trained model, scores new comments.
Also runs rule-based litigation risk detection.
"""
import joblib, re, os
import numpy as np

# Load model once at import time
MODEL_PATH = "model/significance_clf.pkl"
_model = None

def get_model():
    global _model
    if _model is None:
        if os.path.exists(MODEL_PATH):
            _model = joblib.load(MODEL_PATH)
        else:
            raise FileNotFoundError("Model not found. Run: python train.py first.")
    return _model

def score_significance(text: str) -> dict:
    """
    ML significance score using trained TF-IDF + LogisticRegression.
    Returns score 0-100, significant bool, and confidence.
    """
    model = get_model()
    prob = model.predict_proba([text])[0]
    sig_prob = float(prob[1])  # probability of class 1 (significant)
    score = round(sig_prob * 100)
    return {
        "score": score,
        "significant": sig_prob >= 0.5,
        "confidence": "high" if abs(sig_prob - 0.5) > 0.3 else "medium"
    }

# Litigation risk patterns from real APA case law
LIT_PATTERNS = [
    (r'\d+ U\.S\. \d+',                    "Case citation"),
    (r'\d+ U\.S\.C\. §\d+',               "Statute citation"),
    (r'5 U\.S\.C\. §706',                  "APA §706 — judicial review"),
    (r'reserve all rights',                 "Explicit legal threat"),
    (r'seek judicial review',               "Explicit legal threat"),
    (r'arbitrary and capricious',           "A&C challenge"),
    (r'major questions doctrine',           "Major questions doctrine"),
    (r'(Fifth|D\.C\.) Circuit',            "Forum selection signal"),
    (r'injunctive relief',                  "Injunction threat"),
    (r'(LLP|Law Firm|Counsel|Attorney)',    "Law firm author"),
    (r'Executive Order 12866',              "EO 12866 — cost-benefit"),
    (r'Regulatory Flexibility Act|RFA',     "RFA challenge"),
]

def score_litigation(text: str, author: str = "") -> dict:
    """
    Rule-based litigation risk scorer.
    Returns score 0-100, risk level, and list of triggered flags.
    """
    combined = text + " " + author
    triggered = []
    for pattern, label in LIT_PATTERNS:
        if re.search(pattern, combined, re.IGNORECASE):
            triggered.append(label)

    score = min(100, round(len(triggered) / len(LIT_PATTERNS) * 100))
    risk = "high" if score >= 50 else "medium" if score >= 25 else "low"
    return {
        "score": score,
        "risk": risk,
        "flags": triggered
    }

def detect_form_letter(comments: list) -> list:
    """
    Detect coordinated form letter campaigns by text similarity.
    Groups near-identical comments together.
    """
    from collections import defaultdict
    groups = defaultdict(list)
    for c in comments:
        # Use first 80 chars as fingerprint
        key = re.sub(r'\s+', ' ', c["text"][:80].lower().strip())
        groups[key].append(c["id"])

    campaign_map = {}
    for key, ids in groups.items():
        if len(ids) > 1:
            for cid in ids:
                campaign_map[cid] = {"campaign_size": len(ids), "is_form_letter": True}

    result = []
    for c in comments:
        c["form_letter"] = campaign_map.get(c["id"], {"is_form_letter": False})
        result.append(c)
    return result

def extract_cfr_sections(text: str) -> list:
    """Extract CFR section references from comment text."""
    pattern = r'§\s*[\d]+\.[\d]+[\w\-]*'
    matches = re.findall(pattern, text)
    return list(set(matches)) if matches else ["General"]

def score_comment(comment: dict) -> dict:
    """Run full scoring pipeline on a single comment."""
    text = comment.get("text", "")
    author = comment.get("name", "")
    sig = score_significance(text)
    lit = score_litigation(text, author)
    cfr = extract_cfr_sections(text)
    return {
        **comment,
        "significance_score": sig["score"],
        "significant": sig["significant"],
        "sig_confidence": sig["confidence"],
        "litigation_score": lit["score"],
        "litigation_risk": lit["risk"],
        "litigation_flags": lit["flags"],
        "cfr_sections": cfr,
    }
```

## Step 3 — response_gen.py
```python
"""
Claude API — only used for language generation tasks:
1. Name theme clusters
2. Draft APA preamble responses
3. Generate OGC memo
4. Plain-language policy summary
"""
import anthropic, json

client = anthropic.Anthropic()  # reads ANTHROPIC_API_KEY from environment

def call_claude(prompt: str, max_tokens: int = 1000) -> str:
    response = client.messages.create(
        model="claude-sonnet-4-20250514",
        max_tokens=max_tokens,
        messages=[{"role": "user", "content": prompt}]
    )
    return response.content[0].text

def summarize_policy(title: str, text: str) -> str:
    """Plain-language summary of the proposed rule."""
    prompt = f"""Summarize this proposed federal rule in 2 sentences at grade 8 reading level.
Rule: {title}
Text: {text[:500]}
Respond with just the summary, no preamble."""
    return call_claude(prompt, max_tokens=200)

def cluster_and_name(comment_texts: list, policy_title: str) -> list:
    """
    Simple clustering: group comments by keyword similarity,
    then use Claude to name each cluster.
    Returns list of clusters with names and comment indices.
    """
    from sklearn.feature_extraction.text import TfidfVectorizer
    from sklearn.cluster import KMeans
    import numpy as np

    if len(comment_texts) < 3:
        return [{"name": "General comments", "indices": list(range(len(comment_texts)))}]

    n_clusters = min(4, len(comment_texts))
    vectorizer = TfidfVectorizer(max_features=500, stop_words="english")
    X = vectorizer.fit_transform(comment_texts)
    kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
    labels = kmeans.fit_predict(X)

    clusters = {}
    for idx, label in enumerate(labels):
        clusters.setdefault(int(label), []).append(idx)

    # Use Claude to name each cluster
    named = []
    for label, indices in clusters.items():
        sample = " | ".join([comment_texts[i][:100] for i in indices[:3]])
        prompt = f"""These public comments on "{policy_title}" share a theme.
Comments: {sample}
Give this cluster a name in 4-6 words. Respond with ONLY the name."""
        name = call_claude(prompt, max_tokens=20).strip()
        named.append({"name": name, "indices": indices, "count": len(indices)})

    return named

def draft_response(cluster_name: str, comment_samples: list, policy_title: str) -> str:
    """Draft APA preamble response for a theme cluster."""
    samples = "\n".join([f"- {c[:150]}" for c in comment_samples[:3]])
    prompt = f"""You are a federal agency attorney drafting the official Response to Comments
for the final rule preamble of "{policy_title}".

Theme: "{cluster_name}"
Sample comments:
{samples}

Write a 2-paragraph APA-compliant response in Federal Register language.
Paragraph 1: Acknowledge the comments.
Paragraph 2: Agency position and legal/technical rationale.
Be formal and specific."""
    return call_claude(prompt, max_tokens=400)

def generate_ogc_memo(high_risk_comments: list, policy_title: str) -> str:
    """Generate OGC litigation risk memo."""
    comment_list = "\n".join([
        f"- {c['name']}: {', '.join(c['litigation_flags'])} (score: {c['litigation_score']})"
        for c in high_risk_comments[:5]
    ])
    prompt = f"""You are the Office of General Counsel reviewing litigation risk for "{policy_title}".

High-risk comments:
{comment_list}

Write a 2-paragraph internal OGC memo identifying: (1) key litigation risks,
(2) recommended actions for the response team. Use formal legal language."""
    return call_claude(prompt, max_tokens=400)
```

```python
@app.route("/api/process", methods=["POST"])
def process_comments():
    """
    Run full ML pipeline:
    1. ML significance scoring (trained classifier)
    2. Litigation risk scoring (rule-based)
    3. Form letter detection
    4. CFR section extraction
    5. TF-IDF clustering + Claude cluster naming
    6. Claude draft responses per cluster
    """
    comments = session_data.get("comments", [])
    policy = session_data.get("policy", {})

    if not comments:
        return jsonify({"error": "No comments loaded. Call /api/import first."}), 400

    # Step 1–4: ML scoring on each comment
    scored = [score_comment(c) for c in comments]
    scored = detect_form_letter(scored)

    # Step 5: Cluster significant, non-form-letter comments
    sig_comments = [c for c in scored if c["significant"] and not c["form_letter"]["is_form_letter"]]
    texts = [c["text"] for c in sig_comments]

    try:
        clusters = cluster_and_name(texts, policy.get("title", ""))
    except Exception as e:
        clusters = [{"name": "Substantive Comments", "indices": list(range(len(texts))), "count": len(texts)}]

    # Step 6: Draft Claude response for each cluster
    full_clusters = []
    for cl in clusters:
        samples = [texts[i] for i in cl["indices"][:3]]
        comment_ids = [sig_comments[i]["id"] for i in cl["indices"]]
        try:
            draft = draft_response(cl["name"], samples, policy.get("title", ""))
        except Exception:
            draft = f"The agency has carefully considered comments on {cl['name']} and responds as follows..."
        full_clusters.append({**cl, "comment_ids": comment_ids, "draft_response": draft})

    # OGC memo for high-risk comments
    high_risk = [c for c in scored if c["litigation_risk"] == "high"]
    try:
        ogc_memo = generate_ogc_memo(high_risk, policy.get("title", "")) if high_risk else "No high-risk comments identified."
    except Exception:
        ogc_memo = "OGC review pending."

    session_data["scored_comments"] = scored
    session_data["clusters"] = full_clusters
    session_data["ogc_memo"] = ogc_memo

    # Stats
    stats = {
        "total": len(scored),
        "significant": sum(1 for c in scored if c["significant"]),
        "form_letters": sum(1 for c in scored if c["form_letter"]["is_form_letter"]),
        "litigation_high": len(high_risk),
        "clusters": len(full_clusters),
    }

    return jsonify({
        "stats": stats,
        "comments": scored,
        "clusters": full_clusters,
        "ogc_memo": ogc_memo,
    })

    })


if __name__ == "__main__":
    print("PolicyPulse AI — Flask server starting...")
    print("Health check: http://localhost:5000/api/health")
    app.run(debug=True, port=5000)
```
